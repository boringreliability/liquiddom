**Corrections to the README "Shared interfaces" found against the code (applied in this plan; carried into README and W75 by the cross-ward verification 2026-10-10, see `_verification.md`):**
1. **The renderer needs the atlas pixels, not only the rects.** The README gives `ElementPaint.atlasRect` only. 2. **`TextAtlas` needs `version`, `dpr`, `stale` and a `maxSize` option.** `version` bumps per built canvas (the renderer re-uploads on change), `dpr` maps atlas px back to CSS px, `stale` is what `invalidate(id)` sets for W75 (the runtime rebuilds while it is true), `maxSize` (default `ATLAS_MAX_SIZE_PX = 4096`) makes Review Focus 4 testable. `build(entries)` keeps the README signature; every entry carries the same `dpr` (the runtime passes `viewport.dpr`), the first one wins.

//! W74.0 throwaway (never merged): re-form with the F rest condition, F residual under

#![allow(clippy::indexing_slicing)]

use super::api::FluidCore;
use super::scenario_tests::{LAYOUT, SEED, acceptance_core};

const SPLASH: usize = 0;
const CARD: usize = 3;

fn frame(c: &mut FluidCore) {
    c.tick(1.0 / 60.0, -1.0e4, -1.0e4, 0.0, 0.0, false, 0.0, 0.0);
}

fn centre(id: usize) -> (f32, f32) {
    let [x, y, w, h, _] = LAYOUT[id];
    (x + 0.5 * w, y + 0.5 * h)
}

fn settled() -> FluidCore {
    let mut c = acceptance_core(SEED);
    for _ in 0..60 {
        frame(&mut c);
    }
    c
}

fn alphas(c: &FluidCore) -> [f32; 4] {
    [0, 1, 2, 3].map(|id| c.state(id)[2])
}

fn slot3(c: &FluidCore) -> [f32; 4] {
    [0, 1, 2, 3].map(|id| c.state(id)[3])
}

            c.splash(SPLASH as u32, x, y, 1.0);

        ("splash 2 on the card", |c| {
            let (x, y) = centre(CARD);
            c.splash(CARD as u32, x, y, 2.0);
        }),
        ("shake 1", |c| c.shake(1.0)),
    ];
    for (name, act) in acts {
        let mut c = settled();
        act(&mut c);
        let mut worst_while_moving = [0.0f32; 4];

        for f in 1..=600u32 {

            let (a, d) = (alphas(&c), slot3(&c));

                if a[id] < 1.0 {

                }

            if f % 20 == 0 {

            if a.iter().all(|&v| v >= 1.0) {
                rest_frame = Some(f);
                break;

        trace.push(c.state(SPLASH)[2]);

    /// W74 (D74-1): F statistics over the homed particles (σ from the eigenvalues of FᵀF,
    /// independent of `fmath::svd2`).
    pub(crate) fn f_stats(&self) -> FStats {
        let p = &self.particles;
        let mut s = FStats::NONE;
        for i in 0..p.cap {
            if p.home_of(i).is_none() {
                continue;
            }
            let [a, b, c, d] = p.f[i];
            if !(a.is_finite() && b.is_finite() && c.is_finite() && d.is_finite()) {
                s.non_finite += 1;
                continue;
            }
            let (p00, p01, p11) = (a * a + c * c, a * b + c * d, b * b + d * d);
            let half_tr = 0.5 * (p00 + p11);
            let disc = (half_tr * half_tr - (p00 * p11 - p01 * p01)).max(0.0).sqrt();
            s.min_det = s.min_det.min(a * d - b * c);
            s.max_sigma = s.max_sigma.max((half_tr + disc).sqrt());
            s.min_sigma = s.min_sigma.min((half_tr - disc).max(0.0).sqrt());
            if p.flags[i] & super::layout::FLAG_TORN != 0 {
                s.torn += 1;
            }
        }
        s
    }

/** Stubs Range.prototype.getClientRects with `layout(node, start, end)`; returns the restore function. */
export function installRangeRects(layout: (node: Text, start: number, end: number) => DOMRect[]): () => void {
  const proto = Range.prototype as unknown as { getClientRects?: unknown };
  const had = Object.prototype.hasOwnProperty.call(proto, "getClientRects");
  const previous = proto.getClientRects;
  proto.getClientRects = function getClientRects(this: Range): DOMRect[] {
    return layout(this.startContainer as Text, this.startOffset, this.endOffset);
  };
  return () => {
    if (had) proto.getClientRects = previous;
    else delete proto.getClientRects;
  };
}

    const at = lineOf(node);
    return [rect(at.left + start * charW, at.top, (end - start) * charW, lineH)];

async function canvasTextPixelsIn(page: Page, selector: string): Promise<number> {
  return page.evaluate(
    ([sel, min]) => {

      let n = 0;
      for (let py = Math.floor((b.top - cr.top) * sy); py < Math.ceil((b.bottom - cr.top) * sy); py++) {

/** Decodes two PNG screenshots in the page: mean |Δrgb| / 255 and the text energy of each. */
async function screenshotDiff(page: Page, a: Buffer, b: Buffer): Promise<{ mae: number; textA: number; textB: number }> {
  return page.evaluate(
    async ([pa, pb, floor]) => {
      const load = async (b64: string): Promise<ImageData> => {
        const blob = await (await fetch(`data:image/png;base64,${b64}`)).blob();
        const bmp = await createImageBitmap(blob);
        const c = new OffscreenCanvas(bmp.width, bmp.height);
        const ctx = c.getContext("2d");
        if (!ctx) throw new Error("W74: no 2d context for the screenshot diff");
        ctx.drawImage(bmp, 0, 0);
        return ctx.getImageData(0, 0, bmp.width, bmp.height);
      };
      const A = await load(pa);
      const B = await load(pb);
      const n = Math.min(A.data.length, B.data.length);
      let sum = 0;
      let textA = 0;
      let textB = 0;
      for (let i = 0; i < n; i += 4) {
        for (let k = 0; k < 3; k++) sum += Math.abs(A.data[i + k] - B.data[i + k]);
        textA += Math.max(0, Math.min(A.data[i], A.data[i + 1], A.data[i + 2]) - floor);
        textB += Math.max(0, Math.min(B.data[i], B.data[i + 1], B.data[i + 2]) - floor);
      }
      return { mae: sum / ((n / 4) * 3 * 255), textA, textB };
    },
    [a.toString("base64"), b.toString("base64"), TEXT_ENERGY_FLOOR] as const,
  );
}

//! W74 (D74-1, D74-3): the render-only deformation gradient F (spec §2 "Deformation
//! gradient F"): a closed-form 2×2 SVD, the σ clamp with det < 0 flipped to the nearest
//! proper matrix, and the per-substep update with the sticky `torn` hysteresis. Pure,
//! allocation-free and panic-free. Matrices are row-major `[m00, m01, m10, m11]`, like C.
//!
//! Closed form (J. Blinn, "Consider the lowly 2×2 matrix"): with E = (a + d)/2,
//! F = (a − d)/2, G = (c + b)/2, H = (c − b)/2, Q = √(E² + H²), R = √(F² + G²):
//! M = R(φ)·diag(Q + R, Q − R)·R(θ), φ = (atan2(H, E) + atan2(G, F))/2,
//! θ = (atan2(H, E) − atan2(G, F))/2, R(t) the rotation by t. Q − R < 0 ⇔ det M < 0.

use super::particles::IDENTITY;
use super::solver::SLIP_RATE_PER_S;

/// Spec §2: singular values outside [F_SIGMA_MIN, F_SIGMA_MAX] are clamped and tear the particle.
pub const F_SIGMA_MIN: f32 = 0.2;
pub const F_SIGMA_MAX: f32 = 5.0;
/// D74-3: a torn particle heals only once both singular values are back in this band.
pub const TORN_CLEAR_MIN: f32 = 0.67;
pub const TORN_CLEAR_MAX: f32 = 1.5;
/// D74-2: rest also needs, per element, max ‖F − I‖_F below this.
pub const F_REST_TOL: f32 = 0.05;
/// D74-2 (amended after W74.0, saga dec_f34db06a): F's own slip rate towards I, 1.5 × the
/// liquid's SLIP_RATE_PER_S, so F reaches F_REST_TOL no later than the positions reach rest
/// (the shared rate gave the strength-2 card splash 218 frames, budget 180). Only `update_f` uses it.
pub const F_SLIP_RATE_PER_S: f32 = 1.5 * SLIP_RATE_PER_S;

/// 2×2 product `m · n`, row-major.
#[inline]
pub fn mul2(m: [f32; 4], n: [f32; 4]) -> [f32; 4] {
    let [a, b, c, d] = m;
    let [e, f, g, h] = n;
    [a * e + b * g, a * f + b * h, c * e + d * g, c * f + d * h]
}

/// Blinn's terms of `m`: (E, F, G, H, Q, R). `sqrt` instead of `hypot`: F stays within
/// σ ∈ [0.2, 5], far from overflow, and this runs per particle per substep.
#[inline]
fn blinn(m: [f32; 4]) -> (f32, f32, f32, f32, f32, f32) {
    let [a, b, c, d] = m;
    let e = 0.5 * (a + d);
    let f = 0.5 * (a - d);
    let g = 0.5 * (c + b);
    let h = 0.5 * (c - b);
    (e, f, g, h, (e * e + h * h).sqrt(), (f * f + g * g).sqrt())
}

/// (φ, θ) of the closed form.
#[inline]
fn angles(e: f32, f: f32, g: f32, h: f32) -> (f32, f32) {
    let (a1, a2) = (g.atan2(f), h.atan2(e));
    (0.5 * (a2 + a1), 0.5 * (a2 - a1))
}

/// R(t) = [[cos t, −sin t], [sin t, cos t]].
#[inline]
fn rot(t: f32) -> [f32; 4] {
    let (s, c) = t.sin_cos();
    [c, -s, s, c]
}

/// Closed-form 2×2 SVD: `F = U·diag(σ)·Vᵀ`, σ = [σ1, σ2] with σ1 ≥ σ2 ≥ 0. U is a rotation;
/// V is a rotation when det F ≥ 0 and a reflection when det F < 0. No allocation.
pub fn svd2(f: [f32; 4]) -> ([f32; 4], [f32; 2], [f32; 4]) {
    let (e, ff, g, h, q, r) = blinn(f);
    let (phi, theta) = angles(e, ff, g, h);
    let sy = q - r;
    // Vᵀ = diag(1, sign(sy))·R(θ), so U·diag(σ1, |sy|)·Vᵀ = R(φ)·diag(σ1, sy)·R(θ) = F.
    let [t0, t1, t2, t3] = rot(theta);
    let s = if sy < 0.0 { -1.0 } else { 1.0 };
    (rot(phi), [q + r, sy.abs()], [t0, s * t2, t1, s * t3])
}

/// Spec §2 safety clamp: F is returned unchanged when σ1 ≤ F_SIGMA_MAX and the signed
/// σ2 ≥ F_SIGMA_MIN (which implies det F > 0). Otherwise both singular values are clamped into
/// the band and a negative det is flipped to the nearest proper matrix (σ2 changes sign);
/// `hit_clamp` is true.
pub fn clamp_f(f: [f32; 4]) -> ([f32; 4], bool) {
    let (e, ff, g, h, q, r) = blinn(f);
    let (sx, sy) = (q + r, q - r);
    if sx <= F_SIGMA_MAX && sy >= F_SIGMA_MIN {
        return (f, false);
    }
    (clamped(e, ff, g, h, sx, sy), true)
}

/// `R(φ)·diag(clamp(sx), clamp(|sy|))·R(θ)` from Blinn's terms without trigonometry (a clamped
/// particle is re-clamped on every substep, so this is hot): F is the sum of its conformal part
/// Q = [[E, −H], [H, E]] (= q·R(φ+θ)) and its anticonformal part A = [[F, G], [G, −F]]
/// (= r·R(φ)·diag(1, −1)·R(θ)), so scaling them by q'/q and r'/r with q' = (σ1' + σ2')/2 and
/// r' = (σ1' − σ2')/2 gives the clamped matrix. Only q ≈ 0 (a pure reflection-scale, where
/// R(φ+θ) is undefined) takes the angle path. Plan validation 2026-10-10: the native F-cost
/// ratio fell from 1.31 (angles) to 1.22.
#[inline]
fn clamped(e: f32, ff: f32, g: f32, h: f32, sx: f32, sy: f32) -> [f32; 4] {
    let s1 = sx.clamp(F_SIGMA_MIN, F_SIGMA_MAX);
    let s2 = sy.abs().clamp(F_SIGMA_MIN, F_SIGMA_MAX);
    let q = 0.5 * (sx + sy);
    let r = 0.5 * (sx - sy);
    if q <= 1e-6 {
        let (phi, theta) = angles(e, ff, g, h);
        return mul2(mul2(rot(phi), [s1, 0.0, 0.0, s2]), rot(theta));
    }
    let kq = 0.5 * (s1 + s2) / q;
    let ka = if r > 1e-6 { 0.5 * (s1 - s2) / r } else { 0.0 };
    [kq * e + ka * ff, ka * g - kq * h, kq * h + ka * g, kq * e - ka * ff]
}

/// ‖F − I‖ (Frobenius), D74-2's rest measure.
#[inline]
pub fn f_dev(f: [f32; 4]) -> f32 {
    let [a, b, c, d] = f;
    ((a - 1.0) * (a - 1.0) + b * b + c * c + (d - 1.0) * (d - 1.0)).sqrt()
}

/// One substep of the render-only F (spec §2, D74-1, D74-3): `F ← (I + dt·C)·F`, the slip
/// `F ← F + k·(I − F)` with `k = clamp(dt·F_SLIP_RATE_PER_S·s², 0, 1)` (s clamped to [0, 1],
/// non-finite → 1), then `clamp_f`. `torn` is sticky: set when the clamp hits, cleared only
/// when both singular values of the new F lie in [TORN_CLEAR_MIN, TORN_CLEAR_MAX]. A
/// non-finite intermediate F (from F, C or dt) gives `(I, true)`.
pub fn update_f(f: [f32; 4], c: [f32; 4], dt: f32, stiffness: f32, torn: bool) -> ([f32; 4], bool) {
    let [c00, c01, c10, c11] = c;
    let mut g = mul2([1.0 + dt * c00, dt * c01, dt * c10, 1.0 + dt * c11], f);
    let s = if stiffness.is_finite() { stiffness.clamp(0.0, 1.0) } else { 1.0 };
    let k = (dt * F_SLIP_RATE_PER_S * s * s).clamp(0.0, 1.0);
    for (gi, id) in g.iter_mut().zip(IDENTITY) {
        *gi += k * (id - *gi);
    }
    if !g.iter().all(|v| v.is_finite()) {
        return (IDENTITY, true);
    }
    // `clamp_f` inlined so Blinn's terms are computed once per particle and substep (hot path).
    let (e, ff, gg, h, q, r) = blinn(g);
    let (sx, sy) = (q + r, q - r);
    if sx <= F_SIGMA_MAX && sy >= F_SIGMA_MIN {
        let healed = sy >= TORN_CLEAR_MIN && sx <= TORN_CLEAR_MAX;
        return (g, torn && !healed);
    }
    (clamped(e, ff, gg, h, sx, sy), true)
}

/**
 * W74 (D74-5): the text atlas (spec §3 "Text atlas"; slice-4 design §3). The runtime builds it
 * from the DOM under WebGPU — text nodes only, line boxes from `Range.getClientRects()`, the
 * computed style of each text node's parent, `fillText` word by word at the device pixel ratio,
 * no re-wrapping — into one shelf-packed canvas page with `ceil(10·dpr)` atlas px of gutter
 * around each element. The renderer only uploads the page: rendering never touches the DOM.
 * jsdom has no OffscreenCanvas, so tests inject `createCanvas`.
 *
 * Fail-safe (README Review Focus 1 and 4): an element without text, wider than the page or past
 * its height gets no rect, so it never gets `liquid-text` and its DOM text stays. A factory that
 * returns null (no OffscreenCanvas) or a page without a 2d context gives no atlas, silently; a
 * missing `Range.getClientRects` throws (the runtime warns once and draws no text).
 * W75: `invalidate(id)` marks the page stale; the runtime rebuilds while `stale`.
 */

/** An element's border box in the atlas page, in atlas px (CSS px × dpr). */
export interface AtlasRect {
  readonly x: number;
  readonly y: number;
  readonly w: number;
  readonly h: number;
}

/** Creates the atlas page; null when the environment has no OffscreenCanvas. */
export type AtlasCanvasFactory = (w: number, h: number) => OffscreenCanvas | null;

/** CSS px of empty page around each element (≥ one kernel radius, spec §3): `ceil(10·dpr)` atlas px. */
export const ATLAS_GUTTER_PX = 10;
/** Page edge in atlas px; within WebGPU's guaranteed maxTextureDimension2D (8192). */
export const ATLAS_MAX_SIZE_PX = 4096;

export interface AtlasEntry {
  readonly id: number;
  readonly el: HTMLElement;
  /** Device pixel ratio; every entry of one build carries the same value (the first one wins). */
  readonly dpr: number;
}

export const defaultAtlasCanvas: AtlasCanvasFactory = (w, h) =>
  typeof OffscreenCanvas === "function" ? new OffscreenCanvas(w, h) : null;

/** NodeFilter.SHOW_TEXT, without relying on a NodeFilter global. */
const SHOW_TEXT = 0x4;
const WORD = /\S+/g;

interface Placed {
  readonly id: number;
  readonly el: HTMLElement;
  readonly box: DOMRect;
  readonly x: number;
  readonly y: number;
}

/** True when `el` has a text node with non-whitespace content. */
function hasText(el: HTMLElement): boolean {
  const walker = el.ownerDocument.createTreeWalker(el, SHOW_TEXT);
  for (let n = walker.nextNode(); n; n = walker.nextNode()) {
    if ((n.nodeValue ?? "").trim() !== "") return true;
  }
  return false;
}

function validDpr(dpr: number | undefined): number {
  return dpr !== undefined && Number.isFinite(dpr) && dpr > 0 ? dpr : 1;
}

/** Canvas `font` shorthand from computed style: style, small-caps, weight, size, family. */
function canvasFont(cs: CSSStyleDeclaration): string {
  const caps = cs.fontVariantCaps === "small-caps" ? "small-caps" : "";
  return `${cs.fontStyle} ${caps} ${cs.fontWeight} ${cs.fontSize} ${cs.fontFamily}`.replace(/\s+/g, " ").trim();
}

/** The DOM draws transformed glyphs; the word is transformed the same way (capitalize per word). */
function transformed(word: string, textTransform: string): string {
  if (textTransform === "uppercase") return word.toUpperCase();
  if (textTransform === "lowercase") return word.toLowerCase();
  if (textTransform === "capitalize") return word.charAt(0).toUpperCase() + word.slice(1);
  return word;
}

function finiteOr(v: number | undefined, fallback: number): number {
  return v !== undefined && Number.isFinite(v) ? v : fallback;
}

/** Fills every word of `el`'s text nodes at its first line box: client CSS px mapped onto the page. */
function drawText(ctx: OffscreenCanvasRenderingContext2D, el: HTMLElement, box: DOMRect, ox: number, oy: number, dpr: number): void {
  const doc = el.ownerDocument;
  const view = doc.defaultView;
  if (!view) return;
  const range = doc.createRange();
  ctx.setTransform(dpr, 0, 0, dpr, ox - box.left * dpr, oy - box.top * dpr);
  ctx.textBaseline = "alphabetic";
  const walker = doc.createTreeWalker(el, SHOW_TEXT);
  for (let n = walker.nextNode(); n; n = walker.nextNode()) {
    const text = n.nodeValue ?? "";
    const parent = n.parentElement;
    if (!parent || text.trim() === "") continue;
    const cs = view.getComputedStyle(parent);
    if (cs.visibility === "hidden" || cs.visibility === "collapse") continue;
    ctx.font = canvasFont(cs);
    ctx.fillStyle = cs.color;
    if ("letterSpacing" in ctx) {
      (ctx as { letterSpacing: string }).letterSpacing = cs.letterSpacing && cs.letterSpacing !== "normal" ? cs.letterSpacing : "0px";
    }
    const size = parseFloat(cs.fontSize) || 16;
    for (const m of text.matchAll(WORD)) {
      const start = m.index ?? 0;
      range.setStart(n, start);
      range.setEnd(n, start + m[0].length);
      const line = range.getClientRects()[0];
      if (!line || !(line.width > 0 && line.height > 0)) continue;
      const word = transformed(m[0], cs.textTransform);
      const tm = ctx.measureText(word);
      const ascent = finiteOr(tm.fontBoundingBoxAscent, 0.8 * size);
      const descent = finiteOr(tm.fontBoundingBoxDescent, 0.2 * size);
      // The line box centres the font box: baseline = top + (height − (ascent + descent)) / 2 + ascent.
      ctx.fillText(word, line.left, line.top + (line.height - (ascent + descent)) / 2 + ascent);
    }
  }
  ctx.setTransform(1, 0, 0, 1, 0, 0);
}

export class TextAtlas {
  private readonly createCanvas: AtlasCanvasFactory;
  private readonly maxSize: number;
  private page: OffscreenCanvas | null = null;
  private pageW = 0;
  private pageH = 0;
  private pageDpr = 1;
  private builds = 0;
  private dirty = false;

  constructor(opts: { createCanvas?: AtlasCanvasFactory; maxSize?: number } = {}) {
    const max = opts.maxSize ?? ATLAS_MAX_SIZE_PX;
    if (!(Number.isInteger(max) && max >= 1)) {
      throw new RangeError(`[liquiddom] TextAtlas maxSize must be a positive integer, got ${String(max)}`);
    }
    this.createCanvas = opts.createCanvas ?? defaultAtlasCanvas;
    this.maxSize = max;
  }

  /** The page of the last successful build; null before, without text or without a canvas. */
  get canvas(): OffscreenCanvas | null {
    return this.page;
  }

  get width(): number {
    return this.pageW;
  }

  get height(): number {
    return this.pageH;
  }

  /** The DPR the page was drawn at: CSS px = atlas px / dpr. */
  get dpr(): number {
    return this.pageDpr;
  }

  /** +1 per built page; the renderer re-uploads when it changes. */
  get version(): number {
    return this.builds;
  }

  /** W75: true after `invalidate()` until the next `build()`. */
  get stale(): boolean {
    return this.dirty;
  }

  /** W75 hook (text, font or DPR change of element `id`); v1 re-renders the whole page on the next build. */
  invalidate(_id: number): void {
    this.dirty = true;
  }

  /**
   * Packs every entry with text into one page and draws it. Returns the rect per id; an entry
   * without text, wider than the page or past its height has none. Throws when
   * `Range.getClientRects` is missing (after the page exists; the previous page is kept).
   */
  build(entries: ReadonlyArray<AtlasEntry>): ReadonlyMap<number, AtlasRect> {
    this.dirty = false;
    const rects = new Map<number, AtlasRect>();
    const dpr = validDpr(entries[0]?.dpr);
    const gutter = Math.ceil(ATLAS_GUTTER_PX * dpr);
    const max = this.maxSize;
    const placed: Placed[] = [];
    let x = 0;
    let y = 0;
    let shelf = 0;
    let usedW = 0;
    for (const e of entries) {
      if (!hasText(e.el)) continue;
      const box = e.el.getBoundingClientRect();
      if (!(box.width > 0 && box.height > 0)) continue;
      const slotW = Math.ceil(box.width * dpr) + 2 * gutter;
      const slotH = Math.ceil(box.height * dpr) + 2 * gutter;
      if (slotW > max || slotH > max) continue; // Review Focus 4: larger than the page
      if (x + slotW > max) {
        x = 0;
        y += shelf;
        shelf = 0;
      }
      if (y + slotH > max) continue; // Review Focus 4: the page is full
      placed.push({ id: e.id, el: e.el, box, x, y });
      x += slotW;
      usedW = Math.max(usedW, x);
      shelf = Math.max(shelf, slotH);
    }
    const first = placed[0];
    if (!first) {
      this.clearPage();
      return rects;
    }
    const w = usedW;
    const h = y + shelf;
    const canvas = this.createCanvas(w, h);
    const ctx = canvas?.getContext("2d") ?? null;
    if (!canvas || !ctx) {
      this.clearPage();
      return rects;
    }
    if (typeof first.el.ownerDocument.createRange().getClientRects !== "function") {
      throw new Error("[liquiddom] text atlas: Range.getClientRects() is not available");
    }
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.clearRect(0, 0, w, h);
    for (const p of placed) {
      const ox = p.x + gutter;
      const oy = p.y + gutter;
      drawText(ctx, p.el, p.box, ox, oy, dpr);
      rects.set(p.id, { x: ox, y: oy, w: p.box.width * dpr, h: p.box.height * dpr });
    }
    this.page = canvas;
    this.pageW = w;
    this.pageH = h;
    this.pageDpr = dpr;
    this.builds += 1;
    return rects;
  }

  private clearPage(): void {
    this.page = null;
    this.pageW = 0;
    this.pageH = 0;
  }
}

 * W74 (D74-6): the `liquid-text` toggle (spec §3 "Crisp at rest", §4). The runtime calls
 * `syncTextClasses` once per frame AFTER `renderer.render()`, so the class and the atlas text
 * appear and disappear in the same frame. An element carries the class exactly while the active
 * renderer is WebGPU, motion is not reduced, the renderer draws its atlas text (`textReady`) and
 * its restAlpha < 1. Every other path takes it off (README Review Focus 1): `clearTextClasses`
 * for a lost device, reduced motion, a failed frame, a failed rebuild and destroy;
 * `releaseTextClass` for unobserve. Holds are refcounted across instances in stylesheet.ts;
 * `held` is this instance's own set (one hold per element per instance).
 */
import type { Renderer } from "./renderers/frame";
import type { ActiveRenderer } from "./renderers/select";
import { addTextClass, removeTextClass } from "./stylesheet";

export interface TextToggleArgs {
  /** The observed slot ids this frame. */
  readonly ids: Iterable<number>;
  elementOf(id: number): HTMLElement | undefined;
  restAlpha(id: number): number;
  /** null while a device-lost rebuild is in flight. */
  readonly renderer: Renderer | null;
  readonly activeRenderer: ActiveRenderer;
  readonly reducedMotion: boolean;
  readonly held: Set<HTMLElement>;
}

/** Elements visited by the current sync; reused (the toggle is synchronous and never re-entered). */
const seen = new Set<HTMLElement>();

export function syncTextClasses(a: TextToggleArgs): void {
  seen.clear();
  const r = a.renderer;
  const gate = a.activeRenderer === "webgpu" && !a.reducedMotion && r !== null;
  for (const id of a.ids) {
    const el = a.elementOf(id);
    if (!el) continue;
    seen.add(el);
    const want = gate && r !== null && r.textReady?.(id) === true && a.restAlpha(id) < 1;
    if (want && !a.held.has(el)) {
      a.held.add(el);
      addTextClass(el);
    } else if (!want && a.held.has(el)) {
      a.held.delete(el);
      removeTextClass(el);
    }
  }
  // An element this instance still holds but no longer observes (a missed release) lets go too.
  for (const el of a.held) {
    if (!seen.has(el)) {
      a.held.delete(el);
      removeTextClass(el);
    }
  }
  seen.clear();
}

/** Releases every hold of this instance (fail-safe paths). */
export function clearTextClasses(held: Set<HTMLElement>): void {
  for (const el of held) removeTextClass(el);
  held.clear();
}

/** Releases this instance's hold on `el` (unobserve); no-op when it holds none. */
export function releaseTextClass(held: Set<HTMLElement>, el: HTMLElement): void {
  if (held.delete(el)) removeTextClass(el);
}
